# 数据模型与字段定义

> v0.5 目标设计，应用尚未实施；当前状态见 [执行目录](../execution/README.md)。

## 2. 唯一数据契约

以下是 v0.5 的目标领域形状，尚未改动运行代码。实际实现以唯一 Zod Schema 派生类型与工具参数；这里只定义职责与必要字段，不维护第二套运行协议。Project/Paper 目标 schemaVersion=2，DeckPlan/Deck=3，PlanRecord recordVersion=2；数据库版本与对象版本分开，迁移见[工作流与存储](工作流与存储.md)。

ID 稳定且不含位置含义；PDF 页码从 1 开始，Slide 序号仅用于展示。新增/重排不重建既有身份；revision 单调递增，时间为毫秒，写入身份和基准由应用控制，模型不能分配持久化权限。


字段与关系的合法性见 [数据约束](数据约束.md)；应用接口见 [接口契约](接口契约.md)。

### 项目、全文与来源

```ts
type BBox = { x: number; y: number; width: number; height: number };
type Checkpoint =
  | "project-created" | "pdf-parsed" | "figures-ready" | "paper-ready"
  | "outline-ready" | "deck-plan-ready" | "deck-ready";
type Preferences = {
  instruction: string;
  language?: string;
  targetSlides?: number;
  strategyId?: string;
};
interface Project {
  schemaVersion: 2;
  id: string; name: string; nameIsCustom?: boolean;
  paperId: string; pdfAssetId: string;
  currentDeckId?: string; previousDeckId?: string;
  checkpoint: Checkpoint;
  preferences: Preferences;
  lastOpenedSlideId?: string;
  createdAt: number; updatedAt: number;
}
interface TextBlock {
  id: string;
  pageNumber: number;
  kind: "heading" | "paragraph" | "caption" | "table" | "other";
  text: string;
}
interface SourceReference {
  id: string;
  kind: "text" | "figure" | "panel" | "caption";
  pageNumber: number;
  bbox?: BBox;
  textSpan?: { blockId: string; start: number; end: number };
  textQuote?: string;
  geometryOrigin?: "automatic" | "manual";
}
interface CaptionAssociation {
  links: { sourceId: string; role: "panel" | "shared" }[];
  origin: "automatic" | "manual";
  status: "linked" | "needs-review";
}
interface FigurePanel {
  id: string; label?: string; sourceId: string; description?: string;
  labelOrigin?: "automatic" | "manual";
  captionAssociation?: CaptionAssociation;
}
interface FigureRef {
  id: string; label?: string; caption?: string;
  captionSourceIds?: string[];
  sourceId: string; description?: string; panels: FigurePanel[];
}
```

### 论文底稿与证据关系

```ts
type ClaimStrength = "descriptive" | "associative" | "supportive" | "causal";
interface Claim {
  id: string; text: string; strength: ClaimStrength;
  importance: "primary" | "secondary"; evidenceIds: string[];
}
interface Evidence {
  id: string; kind: string; summary: string; sourceIds: string[];
}
interface StoryPoint { text: string; claimIds: string[]; sourceIds: string[] }
type StoryTopic = "background" | "knowledgeGap" | "question" | "studyDesign"
  | "mainFindings" | "novelty" | "limitations" | "conclusion";
interface FigureSourceSnapshot {
  figures: FigureRef[];
  sources: SourceReference[];
}
interface FigureReview {
  revision: number;
  confirmedRevision?: number;
  confirmedAt?: number;
  automaticBaseline?: FigureSourceSnapshot;
}
interface Paper {
  schemaVersion: 2;
  id: string; projectId: string; pdfAssetId: string; revision: number;
  metadata: { title?: string; authors?: string[]; journal?: string; year?: number; doi?: string };
  pages: { pageNumber: number; width: number; height: number; blockIds: string[] }[];
  blocks: TextBlock[];
  sources: SourceReference[];
  figures: FigureRef[];
  figureReview: FigureReview;
  pendingEvidenceFigureIds: string[];
  studyProfile?: { type: string; designSummary: string; sourceIds: string[] };
  story?: Record<StoryTopic, StoryPoint[]>;
  claims: Claim[];
  evidences: Evidence[];
}
```

### 大纲、演讲稿与文稿

```ts
type LayoutId = "title" | "text-only" | "figure-full" | "figure-text" | "two-figures" | "panel-grid";
type SlideKind = "title" | "background" | "question" | "method" | "result"
  | "summary" | "discussion" | "conclusion" | "custom";
type SectionKind = "opening" | "background" | "question" | "study-design"
  | "results" | "synthesis" | "limitations" | "takeaways" | "discussion" | "custom";
type ContentTrack = "main" | "supplement";
interface DeckSection {
  id: string; kind: SectionKind; track: ContentTrack;
  title: string; purpose: string; transitionToNext?: string;
}
interface PlannedFigure { figureId: string; panelId?: string }
interface SpeechSegment {
  id: string; sectionId: string;
  purpose: string; text: string;
  claimIds: string[]; sourceIds: string[];
}
interface PlannedSlide {
  id: string; sectionId: string; kind: SlideKind;
  title: string; purpose: string; message: string;
  layoutId: LayoutId; speechIds: string[];
  claimIds: string[]; sourceIds: string[]; figures: PlannedFigure[];
}
type SlideElement =
  | { id: string; type: "text"; text: string }
  | { id: string; type: "bullet-list"; items: string[] }
  | { id: string; type: "figure"; figureId: string; panelId?: string; cropOverride?: BBox }
  | { id: string; type: "citation"; sourceIds: string[] };
interface Slide {
  id: string; sectionId: string; kind: SlideKind; title: string;
  purpose?: string; message?: string; layoutId: LayoutId;
  speechIds: string[]; elements: SlideElement[];
  claimIds: string[]; sourceIds: string[];
}
interface CoverageOmission {
  claimId: string;
  reason: string;
  origin: "user";
}
interface DeckPlan {
  schemaVersion: 3;
  id: string; paperId: string; paperRevision: number;
  revision: number; status: "draft" | "ready";
  title: string; language: string;
  sections: DeckSection[]; speech: SpeechSegment[];
  slides: PlannedSlide[]; omissions: CoverageOmission[];
  createdAt: number; updatedAt: number;
}
interface Deck {
  schemaVersion: 3;
  id: string; paperId: string; paperRevision: number; revision: number;
  title: string; language: string;
  sections: DeckSection[]; speech: SpeechSegment[]; slides: Slide[];
  omissions: CoverageOmission[];
  createdAt: number; updatedAt: number;
}
```
